# stay-kit Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extract the shipped Bluebill site into a shared `stay-kit` package, migrate Bluebill onto it with proven parity, then build 243 Lincoln Ave (three units) from a content file alone.

**Architecture:** `stay-kit` is a source package consumed by git ref, holding design tokens, blocks, nine opt-in modules, the content types, and the verification harness. A property repo holds only content, photographs, an accent colour and a mark. Units are an array, so one unit and three cost the same. Credentials are an opaque type that cannot be rendered as JSX, and a build check proves no secret reaches public HTML.

**Tech Stack:** Next.js 16 App Router, React 19, TypeScript strict, plain CSS custom properties, Vitest, Playwright + axe for the harness, Upstash Redis for holds, Resend for mail, pnpm.

**Spec:** `~/Claude/stay-kit/docs/superpowers/specs/2026-08-16-stay-kit-design.md`

## Global Constraints

1) **pnpm only.** Never npm or yarn.
2) **Banned word.** One word must never appear in any output. Always "Vanderbilt Surf" or "the association". `scripts/check-banned-word.mjs` moves into the kit and fails the build on anything new.
3) **No em dashes (U+2014)** in any client-facing string.
4) **No invented facts.** Unknown values use the `UNKNOWN` sentinel and render as a visible marked slot.
5) **Dates render through the property's timezone.** Never build a date from a bare string.
6) **A credential can never reach a public route.** Enforced by an opaque type plus a build check that greps served HTML.
7) **No silent no-ops.** An unconfigured integration must look different on screen from a working one.
8) **Server Components by default.** `'use client'` only where interactivity requires it.
9) **Bluebill is live in production.** All migration work happens on a branch and merges only after the harness proves parity.

```ts
/** A fact nobody has recorded. Renders as a visible marked slot, never a guess. */
export const UNKNOWN = Symbol("unknown-fact");
export type Fact = string | typeof UNKNOWN;

const SECRET_BRAND = Symbol("secret");

/**
 * An opaque box around a credential.
 *
 * It is deliberately NOT a string: React cannot render it, template literals
 * cannot interpolate it, and both toString and toJSON are overridden so it
 * cannot leak by accident. Only `revealSecret` opens it, and only the arrival
 * module calls that. `harness/secrets.mjs` proves the guarantee at build time
 * by fetching every public route and asserting no declared secret appears.
 */
export type Secret = {
  readonly [SECRET_BRAND]: true;
  toString(): string;
  toJSON(): string;
};

export function secret(value: string): Secret {
  return {
    [SECRET_BRAND]: true as const,
    toString: () => "[secret]",
    toJSON: () => "[secret]",
    // Non-enumerable so it survives neither spread nor JSON.
    ...Object.defineProperty({}, "__v", { value, enumerable: false }),
    __read: () => value,
  } as unknown as Secret;
}

export function isSecret(v: unknown): v is Secret {
  return typeof v === "object" && v !== null && SECRET_BRAND in v;
}

export function revealSecret(s: Secret): string {
  return (s as unknown as { __read: () => string }).__read();
}

export type Operator = {
  name: string;
  url?: string;
  phone: string;
  email: string;
};

export type Money = {
  amount: number;
```

```ts
import { isSecret, type Property } from "./types";

const ISO = /^\d{4}-\d{2}-\d{2}$/;
const EM_DASH = "\u2014"; // the character itself, written as an escape

function validTimezone(tz: string): boolean {
  try {
    new Intl.DateTimeFormat("en", { timeZone: tz });
    return true;
  } catch {
    return false;
  }
}

/** Walks every string in a value looking for an em dash. */
function hasEmDash(v: unknown): boolean {
  if (typeof v === "string") return v.includes(EM_DASH);
  if (Array.isArray(v)) return v.some(hasEmDash);
  if (v && typeof v === "object" && !isSecret(v)) return Object.values(v).some(hasEmDash);
  return false;
}

export function validateProperty(p: Property): string[] {
  const problems: string[] = [];

  if (!validTimezone(p.timezone)) {
    problems.push(`timezone: "${p.timezone}" is not a valid IANA timezone`);
  }

  if (p.units.length === 0) {
    problems.push("units: at least one unit is required");
  }

  const seen = new Set<string>();
  for (const u of p.units) {
    if (seen.has(u.id)) problems.push(`units: duplicate unit id "${u.id}"`);
    seen.add(u.id);
    if (u.status.kind === "availableFrom" && !ISO.test(u.status.date)) {
      problems.push(`units[${u.id}]: status date "${u.status.date}" is not YYYY-MM-DD`);
    }
    if (u.status.kind === "held" && !ISO.test(u.status.until)) {
      problems.push(`units[${u.id}]: status date "${u.status.until}" is not YYYY-MM-DD`);
    }
  }

  if (p.modules.includes("arrival") && !(p.arrival && p.arrival.length)) {
    problems.push("arrival: the module is on but no arrival facts are declared");
  }

  // A credential may only be declared inside `arrival`.
  for (const g of p.guidebook ?? []) {
    for (const f of g.facts ?? []) {
      if (isSecret(f.value)) {
        problems.push(`guidebook[${g.id}]: a secret cannot be declared outside arrival`);
      }
    }
  }

  if (hasEmDash(p.spine)) problems.push("spine: contains an em dash");
  for (const g of p.guidebook ?? []) {
    if (hasEmDash(g)) problems.push(`guidebook[${g.id}]: contains an em dash`);
  }
  for (const r of p.rules ?? []) {
    if (hasEmDash(r)) problems.push(`rules[${r.id}]: contains an em dash`);
  }

  return problems;
}
```

```ts
/**
 * Every date in this kit goes through here. A bare `new Date("2026-08-21")`
 * parses as UTC midnight and renders as the 20th anywhere west of Greenwich,
 * which is how a lease end date silently moves by a day.
 */
export function zonedDateStamp(at: Date, tz: string): string {
  // en-CA formats as YYYY-MM-DD.
  return new Intl.DateTimeFormat("en-CA", {
    timeZone: tz,
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  }).format(at);
}

export function earliestStart(from: Date, leadDays: number, tz: string): string {
  const today = zonedDateStamp(from, tz);
  const [y, m, d] = today.split("-").map(Number);
  // Build in UTC from the already-zoned parts, so the arithmetic cannot drift.
  const base = Date.UTC(y, m - 1, d);
  return new Date(base + leadDays * 86_400_000).toISOString().slice(0, 10);
}

export function formatDate(iso: string, tz: string): string {
  const [y, m, d] = iso.split("-").map(Number);
  return new Intl.DateTimeFormat("en-GB", {
    timeZone: tz,
    day: "numeric",
    month: "long",
    year: "numeric",
  }).format(new Date(Date.UTC(y, m - 1, d, 12)));
}
```

```tsx
import { formatDate } from "../lib/dates";
import type { Unit, UnitStatus } from "../content/types";

function statusLabel(s: UnitStatus, tz: string): { text: string; tone: "open" | "soon" | "closed" } {
  switch (s.kind) {
    case "available":
      return { text: "Available now", tone: "open" };
    case "availableFrom":
      return { text: `Available ${formatDate(s.date, tz)}`, tone: "soon" };
    case "held":
      return { text: `Held until ${formatDate(s.until, tz)}`, tone: "soon" };
    case "leased":
      return { text: s.until ? `Leased to ${formatDate(s.until, tz)}` : "Leased", tone: "closed" };
  }
}

function money(m: Unit["rent"]) {
  return `$${m.amount.toLocaleString("en-US")} a ${m.period}`;
}
```

```ts
export type Env = Record<string, string | undefined>;

/**
 * A hold needs somewhere to live. If Upstash is not configured the module
 * refuses to render rather than showing a button that silently does nothing.
 */
export function holdConfigured(env: Env): boolean {
  return Boolean(env.UPSTASH_REDIS_REST_URL && env.UPSTASH_REDIS_REST_TOKEN);
}

export function holdKey(propertySlug: string, unitId: string): string {
  return `hold:${propertySlug}:${unitId}`;
}

export async function placeHold(
  env: Env,
  propertySlug: string,
  unitId: string,
  email: string,
  hours = 72
): Promise<{ ok: true; until: string } | { ok: false; reason: string }> {
  if (!holdConfigured(env)) return { ok: false, reason: "Holds are not configured on this site." };
  const until = new Date(Date.now() + hours * 3_600_000).toISOString();
  const res = await fetch(
    `${env.UPSTASH_REDIS_REST_URL}/set/${encodeURIComponent(holdKey(propertySlug, unitId))}?EX=${hours * 3600}&NX=true`,
    {
      method: "POST",
      headers: { Authorization: `Bearer ${env.UPSTASH_REDIS_REST_TOKEN}` },
      body: JSON.stringify({ email, until }),
    }
  );
  if (!res.ok) return { ok: false, reason: "We could not place that hold just now." };
  const body = (await res.json()) as { result: string | null };
  if (body.result === null) return { ok: false, reason: "Someone else is holding this unit." };
  return { ok: true, until };
}

export async function readHold(env: Env, propertySlug: string, unitId: string) {
  if (!holdConfigured(env)) return null;
  const res = await fetch(`${env.UPSTASH_REDIS_REST_URL}/get/${encodeURIComponent(holdKey(propertySlug, unitId))}`, {
    headers: { Authorization: `Bearer ${env.UPSTASH_REDIS_REST_TOKEN}` },
  });
  if (!res.ok) return null;
  const body = (await res.json()) as { result: string | null };
  return body.result ? (JSON.parse(body.result) as { email: string; until: string }) : null;
}
```

```ts
export function createHoldRoute(property: Property) {
  return {
    async POST(req: Request) {

          { error: "Holds are not available on this site yet. Please call us." },
          { status: 503 }

      const { unitId, email } = (await req.json()) as { unitId?: string; email?: string };
```

```js
/**
 * Fetches every public route and asserts that no declared secret appears in
 * the served HTML. The type system stops a secret being rendered by accident;
 * this stops it being rendered on purpose.
 */
export async function run({ url, routes, secrets }) {
  const lines = [];
  let pass = true;
  for (const route of routes) {
    const res = await fetch(url + route);
    const html = await res.text();
    for (const s of secrets) {
      if (html.includes(s)) {
        pass = false;
        lines.push(`  LEAK: a declared secret appears in the HTML of ${route}`);
      }
    }
  }
  lines.push(pass ? `  no secret appears on any of ${routes.length} public routes` : "");
  return { pass, lines };
}
```

The site passes its secret values in from an env var at verify time, never from a committed file.
